# Meta-Analysis Skill — R Package Requirements
# Single source of truth: mirrors SKILL.md "Packages" line (15 core packages).
# Packages are NOT auto-installed; user installs manually (see r_engine/setup_packages.R).
# R version required: >= 4.0

# --- Statistical engine (core) ---
metafor          # Statistical engine (rma, rma.mv, escalc, forest, funnel)
meta             # RevMan-compatible (metabin, metacont, metagen, funnel, forest)
netmeta          # Network meta-analysis (NMA, frequentist)
bayesmeta        # Bayesian pairwise random-effects meta

# --- Effect size & robust inference ---
esc              # Effect size conversion (mean->d, d<->g, OR<->logOR, r<->Z)
robumeta         # Robust variance estimation (RVE) for dependent effect sizes
clubSandwich     # Cluster-robust SE, small-sample correction (CR2/HC3)

# --- Advanced modules ---
gemtc            # Bayesian NMA (JAGS backend) — REQUIRES system JAGS preinstalled
dosresmeta       # Dose-response meta
mada             # Diagnostic test accuracy meta (bivariate / SROC)
metagear         # Systematic review workflow (PRISMA, screening)

# --- Graphics / output ---
ggplot2          # Grammar of graphics (base for all plots)
gridExtra        # Arrange multiple plots
forestploter     # Publication-ready forest plots (replaces ggforestplot; R4.6 compatible)
svglite          # SVG vector device — coze/local returns editable SVG
